1. 模型格式简介

常见的 3D 模型格式包括：

  • GLB/GLTF: Khronos 标准格式，二进制或 JSON
  • OBJ: 文本格式，简单易读
  • PLY: 多边形格式，支持颜色
  • STL: 三角形网格格式
// 解析简单的 OBJ 格式 async function parseOBJ(data) { const lines = data.split('\n'); const positions = []; const normals = []; const uvs = []; const indices = []; for (const line of lines) { const parts = line.trim().split(/\s+/); if (parts[0] === 'v') { positions.push(parseFloat(parts[1]), parseFloat(parts[2]), parseFloat(parts[3])); } else if (parts[0] === 'vn') { normals.push(parseFloat(parts[1]), parseFloat(parts[2]), parseFloat(parts[3])); } else if (parts[0] === 'vt') { uvs.push(parseFloat(parts[1]), parseFloat(parts[2])); } else if (parts[0] === 'f') { for (let i = 1; i <= 3; i++) { const idx = parts[i].split('/'); indices.push(parseInt(idx[0]) - 1); } } } return { positions, normals, uvs, indices }; }

2. 创建 GPU 缓冲区

将解析后的模型数据上传到 GPU 缓冲区中，以便渲染时使用。

// 创建顶点和索引缓冲区 function createModelBuffers(device, modelData) { const vertexBuffer = device.createBuffer({ size: Float32Array.BYTES_PER_ELEMENT * modelData.positions.length, usage: GPUBufferUsage.VERTEX | GPUBufferUsage.COPY_DST, mappedAtCreation: true }); new Float32Array(vertexBuffer.getMappedRange()).set(modelData.positions); vertexBuffer.unmap(); const indexBuffer = device.createBuffer({ size: Uint32Array.BYTES_PER_ELEMENT * modelData.indices.length, usage: GPUBufferUsage.INDEX | GPUBufferUsage.COPY_DST, mappedAtCreation: true }); new Uint32Array(indexBuffer.getMappedRange()).set(modelData.indices); indexBuffer.unmap(); return { vertexBuffer, indexBuffer }; }
