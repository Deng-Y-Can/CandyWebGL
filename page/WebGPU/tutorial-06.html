1. 计算着色器基础

计算着色器是 WebGPU 中用于通用 GPU 计算的核心组件。它允许我们利用 GPU 的大规模并行处理能力来加速各种计算任务。

  • Workgroup: 工作组，GPU 并行执行的基本单元
  • Invocation: 工作组内的单个线程
  • Storage Buffer: 可读写的缓冲区

2. 矩阵乘法示例

计算着色器非常适合并行矩阵运算。每个线程可以负责计算结果矩阵的一个元素。

// 矩阵乘法计算着色器 struct Matrix { rows: u32, cols: u32, data: array, }; @group(0) @binding(0) var A: Matrix; @group(0) @binding(1) var B: Matrix; @group(0) @binding(2) var C: Matrix; @compute @workgroup_size(16, 16, 1) fn multiplyMatrices( @builtin(global_invocation_id) id: vec3 ) { let row = id.y; let col = id.x; if (row >= A.rows || col >= B.cols) { return; } var sum: f32 = 0.0; for (var k: u32 = 0; k < A.cols; k++) { sum += A.data[row * A.cols + k] * B.data[k * B.cols + col]; } C.data[row * C.cols + col] = sum; }

3. 图像卷积示例

使用计算着色器进行图像处理，例如边缘检测或模糊效果。

// Sobel 边缘检测卷积 @group(0) @binding(0) var src: texture_2d; @group(0) @binding(1) var dst: texture_storage_2d; const sobelX = array, 3>( array(-1.0, 0.0, 1.0), array(-2.0, 0.0, 2.0), array(-1.0, 0.0, 1.0) ); @compute @workgroup_size(8, 8, 1) fn sobelEdge( @builtin(global_invocation_id) id: vec3 ) { let size = textureDimensions(src); if (id.x >= size.x || id.y >= size.y) { return; } var gx: f32 = 0.0; var gy: f32 = 0.0; for (var dy: i32 = -1; dy <= 1; dy++) { for (var dx: i32 = -1; dx <= 1; dx++) { let px = clamp(i32(id.x) + dx, 0, i32(size.x) - 1); let py = clamp(i32(id.y) + dy, 0, i32(size.y) - 1); let color = textureLoad(src, vec2(px, py), 0); let gray = dot(color.rgb, vec3(0.299, 0.587, 0.114)); gx += gray * sobelX[dy + 1][dx + 1]; gy += gray * sobelX[dx + 1][dy + 1]; } } let edge = sqrt(gx * gx + gy * gy); textureStore(dst, vec2(id.x, id.y), vec4(edge, edge, edge, 1.0)); }
