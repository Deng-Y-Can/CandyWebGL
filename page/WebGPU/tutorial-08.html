1. 帧缓冲概念

帧缓冲（Framebuffer）是一个离屏渲染目标，允许我们先渲染到纹理，然后对纹理进行后处理。

  • 离屏渲染: 渲染到纹理而不是直接到屏幕
  • 后处理: 对渲染结果应用滤镜效果
  • 多重渲染: 可以进行多次后处理
// 创建帧缓冲纹理 const framebufferTexture = device.createTexture({ size: [width, height], format: 'rgba8unorm', usage: GPUTextureUsage.RENDER_ATTACHMENT | GPUTextureUsage.TEXTURE_BINDING }); // 创建深度纹理 const depthTexture = device.createTexture({ size: [width, height], format: 'depth24plus', usage: GPUTextureUsage.RENDER_ATTACHMENT }); // 使用帧缓冲渲染 const pass = encoder.beginRenderPass({ colorAttachments: [{ view: framebufferTexture.createView(), loadOp: 'clear', storeOp: 'store', clearValue: [0, 0, 0, 1] }], depthStencilAttachment: { view: depthTexture.createView(), depthLoadOp: 'clear', depthStoreOp: 'store', depthClearValue: 1.0 } });

2. 高斯模糊效果

高斯模糊是一种常用的平滑效果，使用高斯函数计算权重进行像素混合。

// 高斯模糊片段着色器 @fragment fn main(@location(0) uv: vec2) -> @location(0) vec4 { var color: vec4 = vec4(0.0); const blurRadius: f32 = 3.0; const samples: i32 = 7; let step = blurRadius / f32(samples); var weightSum: f32 = 0.0; for (var i: i32 = -samples; i <= samples; i++) { for (var j: i32 = -samples; j <= samples; j++) { let offset = vec2(f32(i) * step, f32(j) * step); let gaussian = exp(-(offset.x * offset.x + offset.y * offset.y) / (2.0 * blurRadius)); color += textureSample(texture, sampler, uv + offset) * gaussian; weightSum += gaussian; } } return color / weightSum; }

3. 色调映射

色调映射用于将高动态范围（HDR）颜色转换为可显示的低动态范围（LDR）颜色。

// ACES 色调映射 fn toneMapACES(color: vec3) -> vec3 { const a: f32 = 2.51; const b: f32 = 0.03; const c: f32 = 2.43; const d: f32 = 0.59; const e: f32 = 0.14; return clamp((color * (a * color + b)) / (color * (c * color + d) + e), 0.0, 1.0); } @fragment fn main(@location(0) uv: vec2) -> @location(0) vec4 { let hdrColor = textureSample(texture, sampler, uv).rgb; let ldrColor = toneMapACES(hdrColor); return vec4(ldrColor, 1.0); }
